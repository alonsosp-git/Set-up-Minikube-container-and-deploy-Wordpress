# ================================================================================
#  MYSQL PERSISTENT VOLUME + CLAIM
#  Stores MySQL data at /data/mysql inside Minikube — survives pod restarts
# ================================================================================

@"
apiVersion: v1
kind: PersistentVolume
metadata:
  name: mysql-pv
spec:
  capacity:
    storage: 1Gi
  accessModes:
    - ReadWriteOnce
  persistentVolumeReclaimPolicy: Retain
  storageClassName: manual
  hostPath:
    path: "/data/mysql"
"@ | Out-File -FilePath "$env:USERPROFILE\k8s\mysql-pv.yaml" -Encoding utf8

kubectl apply -f "$env:USERPROFILE\k8s\mysql-pv.yaml"

@"
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: mysql-pvc
spec:
  accessModes:
    - ReadWriteOnce
  resources:
    requests:
      storage: 1Gi
  storageClassName: manual
"@ | Out-File -FilePath "$env:USERPROFILE\k8s\mysql-pvc.yaml" -Encoding utf8

kubectl apply -f "$env:USERPROFILE\k8s\mysql-pvc.yaml"

  persistentVolumeReclaimPolicy: Retain

  hostPath:
    path: "/data/wordpress"
"@ | Out-File -FilePath "$env:USERPROFILE\k8s\wordpress-pv.yaml" -Encoding utf8

kubectl apply -f "$env:USERPROFILE\k8s\wordpress-pv.yaml"

@"
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: wordpress-pvc
spec:
  accessModes:
    - ReadWriteOnce
  resources:
    requests:
      storage: 2Gi
  storageClassName: manual
"@ | Out-File -FilePath "$env:USERPROFILE\k8s\wordpress-pvc.yaml" -Encoding utf8

# ================================================================================
#  MYSQL DEPLOYMENT
#  [FIX #2] livenessProbe added — restarts MySQL if it becomes unresponsive
#  [FIX #3] resource limits added — prevents OOM kill by the kernel
#  [FIX #4] readinessProbe no longer embeds password in the process list
#           (removed -u root -p$MYSQL_ROOT_PASSWORD from mysqladmin ping)
# ================================================================================

@"
apiVersion: apps/v1
kind: Deployment
metadata:
  name: mysql-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: mysql
  template:
    metadata:
      labels:
        app: mysql
    spec:
      containers:
      - name: mysql
        image: mysql:8.0
        ports:
        - containerPort: 3306
        env:
        - name: MYSQL_ROOT_PASSWORD
          valueFrom:
            secretKeyRef:
              name: mysql-secrets
              key: MYSQL_ROOT_PASSWORD
        - name: MYSQL_DATABASE
          value: wordpress
        - name: MYSQL_USER
          value: wpuser
        - name: MYSQL_PASSWORD
          valueFrom:
            secretKeyRef:
              name: mysql-secrets
              key: MYSQL_PASSWORD
        resources:
          requests:
            memory: "256Mi"
            cpu: "100m"
          limits:
            memory: "512Mi"
            cpu: "500m"
        readinessProbe:
          exec:
            command:
            - mysqladmin
            - ping
            - -h
            - "127.0.0.1"
          initialDelaySeconds: 20
          periodSeconds: 10
          failureThreshold: 6
        livenessProbe:
          exec:
            command:
            - mysqladmin
            - ping
            - -h
            - "127.0.0.1"
          initialDelaySeconds: 30
          periodSeconds: 20
          failureThreshold: 3
        volumeMounts:
        - name: mysql-data
          mountPath: /var/lib/mysql
      volumes:
      - name: mysql-data
        persistentVolumeClaim:
          claimName: mysql-pvc
"@ | Out-File -FilePath "$env:USERPROFILE\k8s\mysql-deployment.yaml" -Encoding utf8

# Apply the MySQL Deployment manifest
kubectl apply -f "$env:USERPROFILE\k8s\mysql-deployment.yaml"

# Verify pods are created after applying the deployment
kubectl get pods

# ================================================================================
#  WORDPRESS DEPLOYMENT
#  [FIX #1] volumeMount for /var/www/html added (uses wordpress-pvc)
#  [FIX #2] livenessProbe added — restarts WordPress if it stops responding
#  [FIX #3] resource limits added — prevents OOM kill by the kernel
#  [FIX #5] readinessProbe initialDelaySeconds raised 10->30, failureThreshold
#           added — prevents premature pod restarts during slow startup
# ================================================================================

@"
apiVersion: apps/v1
kind: Deployment
metadata:
  name: wordpress-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: wordpress
  template:
    metadata:
      labels:
        app: wordpress
    spec:
      containers:
      - name: wordpress
        image: wordpress:latest
        ports:
        - containerPort: 80
        env:
        - name: WORDPRESS_DB_HOST
          value: "mysql-service:3306"
        - name: WORDPRESS_DB_NAME
          value: wordpress
        - name: WORDPRESS_DB_USER
          value: wpuser
        - name: WORDPRESS_DB_PASSWORD
          valueFrom:
            secretKeyRef:
              name: mysql-secrets
              key: MYSQL_PASSWORD
        resources:
          requests:
            memory: "128Mi"
            cpu: "100m"
          limits:
            memory: "512Mi"
            cpu: "500m"
        readinessProbe:
          httpGet:
            path: /wp-login.php
            port: 80
          initialDelaySeconds: 30
          periodSeconds: 10
          failureThreshold: 6
        livenessProbe:
          httpGet:
            path: /wp-login.php
            port: 80
          initialDelaySeconds: 60
          periodSeconds: 20
          failureThreshold: 3
        volumeMounts:
        - name: wordpress-data
          mountPath: /var/www/html
      volumes:
      - name: wordpress-data
        persistentVolumeClaim:
          claimName: wordpress-pvc
"@ | Out-File -FilePath "$env:USERPROFILE\k8s\wordpress-deployment.yaml" -Encoding utf8

kubectl apply -f "$env:USERPROFILE\k8s\wordpress-deployment.yaml"

# ================================================================================
#  [FIX #6] INGRESS — catch-all rule (no host: field)
#  IP addresses are not valid Ingress hostnames. The host: field has been removed
#  so the nginx ingress controller accepts all incoming requests regardless of
#  the Host header. Access via http://<minikube-ip>/wp-admin works correctly.
# ================================================================================

$ingressPath = "$env:USERPROFILE\k8s\wordpress-ingress.yaml"

@"
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: wordpress-ingress
  annotations:
    nginx.ingress.kubernetes.io/rewrite-target: /
spec:
  rules:
  - http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: wordpress-service
            port:
              number: 80
"@ | Out-File -FilePath $ingressPath -Encoding utf8

Write-Output "Applying ingress manifest (catch-all, no host restriction)..."
kubectl apply -f $ingressPath

Write-Output "Ingress applied. Describe ingress:"
kubectl describe ingress wordpress-ingress

# ================================================================================
#  OPEN WORDPRESS — tries multiple access methods in priority order
# ================================================================================

function Run-Trim([string]$cmd) {
    $out = Invoke-Expression $cmd 2>$null
    if ($null -eq $out) { return "" }
    return ($out | Out-String).Trim()
}

$minikubeIP = Run-Trim "minikube ip"

# 1) Try minikube service --url (returns a working tunnel/NodePort URL)
$svcUrl = Run-Trim "minikube service --url wordpress-service"
if ($svcUrl -ne "") {
    $openUrl = "$svcUrl/wp-admin/"
    Write-Output "Opening (minikube service --url): $openUrl"
    Start-Process $openUrl
    return
}

# 2) Try ingress catch-all (http://<minikube-ip>/wp-admin)
if ($minikubeIP -ne "") {
    $openUrl = "http://$minikubeIP/wp-admin/"
    Write-Output "Opening (ingress catch-all): $openUrl"
    Start-Process $openUrl
    return
}

# 3) Fallback: minikube ip + NodePort
if ($minikubeIP -ne "") {
    $openUrl = "http://$minikubeIP`:30080/wp-admin/"
    Write-Output "Opening (minikube ip + NodePort fallback): $openUrl"
    Start-Process $openUrl
    return
}

# 4) Final fallback: port-forward to localhost (keeps a new PowerShell window open)
Write-Output "Falling back to kubectl port-forward on localhost:8080 (new window will remain open for forwarding)..."
$pfCommand = "kubectl port-forward svc/wordpress-service 8080:80"
Start-Process powershell -ArgumentList "-NoExit","-Command",$pfCommand
Start-Sleep -Seconds 2
$openUrl = "http://localhost:8080/wp-admin/"
Write-Output "Opening (port-forward): $openUrl"
Start-Process $openUrl
